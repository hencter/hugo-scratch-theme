{{- /* page-nav.html — previous / next within the current section. `.PrevInSection` and `.NextInSection` follow the section's own ordering (weight, then date), which is exactly the order the sidebar shows, so the two navigations never disagree. */ -}} {{- $page := . -}} {{- $ui := site.Params.ui | default dict -}} {{- $show := index $ui "showPrevNext" -}} {{- if eq $show nil }}{{ $show = true }}{{ end -}} {{- $prev := $page.PrevInSection -}} {{- $next := $page.NextInSection -}} {{- if and $show (or $prev $next) -}}
{{- with $prev }} {{ T "previous" }} {{ .LinkTitle }} {{- end }} {{- with $next }} {{ T "next" }} {{ .LinkTitle }} {{- end }}
{{- end -}}
