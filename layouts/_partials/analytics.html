{{- /* analytics.html — opt-in, production-only analytics. Nothing is emitted until a site sets a value: [params.analytics] googleAnalytics = "G-XXXXXXX" plausibleDomain = "example.com" Both scripts are also suppressed outside production, so `hugo server` never pollutes the statistics with your own reloads. */ -}} {{- $analytics := site.Params.analytics | default dict -}} {{- $google := index $analytics "googleAnalytics" -}} {{- $plausible := index $analytics "plausibleDomain" -}} {{- if hugo.IsProduction -}} {{- with $google }} {{- end }} {{- with $plausible }} {{- end }} {{- end -}}
