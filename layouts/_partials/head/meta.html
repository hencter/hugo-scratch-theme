{{- /* head/meta.html — title, description, canonical, robots, authorship. Three rules this partial exists to enforce: 1. The canonical URL is ABSOLUTE. `.Permalink` already is; a hand-written relative path would be silently ignored by crawlers. 2. Paged archives canonicalise to the pager's own URL, not to page 1 — otherwise pages 2..n claim to be duplicates of page 1. 3. Anything that is not a production build is `noindex, nofollow`, so a preview deployment never ends up in a search index. Note: plain `hugo` sets hugo.Environment to "production". Only `hugo server` (and `hugo -e development`) produces the noindex head. */ -}} {{- $page := . -}} {{- $sep := site.Params.titleSeparator | default "·" -}} {{- $seo := site.Params.seo | default dict -}} {{- /* Pagination exists only on these kinds. Probing .Paginator anywhere else (404, robots.txt, the sitemap page, a regular page) is a hard error: "pagination not supported for this page". */ -}} {{- $paginated := in (slice "section" "taxonomy" "term") $page.Kind -}} {{- /* --- Title ------------------------------------------------------- */ -}} {{- $title := "" -}} {{- if $page.IsHome -}} {{- $title = site.Title -}} {{- with site.Params.tagline }}{{ $title = printf "%s %s %s" site.Title $sep . }}{{ end -}} {{- else -}} {{- $title = printf "%s %s %s" $page.Title $sep site.Title -}} {{- end -}} {{- if $paginated -}} {{- with $page.Paginator -}} {{- if gt .PageNumber 1 -}} {{- $title = printf "%s %s %s" $title $sep (T "pageNumber" .) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* --- Description ------------------------------------------------- */ -}} {{- $description := $page.Description -}} {{- if not $description -}} {{- if $page.IsPage -}} {{- $description = $page.Summary -}} {{- else -}} {{- $description = site.Params.description -}} {{- end -}} {{- end -}} {{- $description = $description | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace | truncate 160 -}} {{- /* --- Canonical --------------------------------------------------- */ -}} {{- $canonical := $page.Permalink -}} {{- if $paginated -}} {{- with $page.Paginator -}} {{- if gt .PageNumber 1 }}{{ $canonical = .URL | absURL }}{{ end -}} {{- end -}} {{- end -}} {{- with $page.Params.canonical }}{{ $canonical = . | absURL }}{{ end -}} {{- /* --- Robots ------------------------------------------------------ */ -}} {{- $robots := "index, follow, max-snippet:-1, max-image-preview:large, max-video-preview:-1" -}} {{- if not hugo.IsProduction }}{{ $robots = "noindex, nofollow" }}{{ end -}} {{- if $page.Params.noindex }}{{ $robots = "noindex, nofollow" }}{{ end -}} {{- with $page.Params.robots }}{{ $robots = . }}{{ end -}} {{ $title }} {{ with $description }} {{ end -}} {{- with $page.Params.keywords }} {{ end -}} {{- $author := "" -}} {{- with $page.Params.authors }}{{ $author = delimit . ", " }}{{ else }}{{ with site.Params.author }}{{ $author = . }}{{ end }}{{ end -}} {{- with $author }} {{ end -}} {{- /* article:published_time / article:modified_time are emitted by head/opengraph.html together with the rest of the article metadata. Emitting them here as well would duplicate every tag. */ -}}
