{{- /* pagination.html — paginator control. @context {pager} Paginator @example {{ partial "pagination.html" .Paginator }} The windowed branch exists because an archive with 60 pages must not render 60 links. {{- with $p.Prev -}} {{ T "previous" }} {{- else -}} {{ T "previous" }} {{- end -}} {{- if le $total 8 -}} {{- range $p.Pagers -}} {{- if eq .PageNumber $current -}} {{ .PageNumber }} {{- else -}} {{ .PageNumber }} {{- end -}} {{- end -}} {{- else -}} {{- range $p.Pagers -}} {{- $n := .PageNumber -}} {{- $inWindow := and (ge $n (sub $current $window)) (le $n (add $current $window)) -}} {{- if or (eq $n 1) (eq $n $total) $inWindow -}} {{- if eq $n $current -}} {{ $n }} {{- else -}} {{ $n }} {{- end -}} {{- else if or (eq $n (sub $current (add $window 1))) (eq $n (add $current (add $window 1))) -}} … {{- end -}} {{- end -}} {{- end -}} {{- with $p.Next -}} {{ T "next" }} {{- else -}} {{ T "next" }} {{- end -}}
{{- end -}}
