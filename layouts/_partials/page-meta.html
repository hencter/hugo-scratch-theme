{{- /* page-meta.html — the line under a page title: dates, author, reading time. Two rules that are easy to get wrong: · `.Date` is a time.Time STRUCT, so `{{ with .Date }}` is always true and an undated page would happily print "0001-01-01". Every date here is guarded with `.IsZero`. · Machine-readable dates use with ISO 8601, whatever the visible text looks like. The visible text comes from params.dateFormat, which each language overrides — the localized `:date_long` token falls back to English for locales the data does not cover. */ -}} {{- $page := . -}} {{- $ui := site.Params.ui | default dict -}} {{- $fmt := site.Params.dateFormat | default ":date_long" -}} {{- $showDate := index $ui "showDate" -}}{{- if eq $showDate nil }}{{ $showDate = true }}{{ end -}} {{- $showReading := index $ui "showReadingTime" -}}{{- if eq $showReading nil }}{{ $showReading = true }}{{ end -}} {{- $showWords := index $ui "showWordCount" -}}{{- if eq $showWords nil }}{{ $showWords = false }}{{ end -}} {{- $hasUpdated := false -}} {{- if not $page.Lastmod.IsZero -}} {{- if ne ($page.Lastmod.Format "2006-01-02") ($page.Date.Format "2006-01-02") }}{{ $hasUpdated = true }}{{ end -}} {{- end -}} {{- if or (and $showDate (not $page.Date.IsZero)) $hasUpdated $page.Params.authors (and $showReading $page.IsPage) -}}
    {{- if and $showDate (not $page.Date.IsZero) }}
  • {{ T "published" }} {{ $page.Date | time.Format $fmt }}
    • {{- end -}} {{- if and $showDate $hasUpdated }}
  • {{ T "updated" }} {{ $page.Lastmod | time.Format $fmt }}
    • {{- end -}} {{- with $page.Params.authors }}
  • {{ T "author" }} {{ delimit . ", " }}
    • {{- end -}} {{- if and $showReading $page.IsPage }}
  • {{ T "readingTime" $page.ReadingTime }}
    • {{- end -}} {{- if and $showWords $page.IsPage }}
  • {{ T "wordCount" $page.WordCount }}
    • {{- end -}} {{- with $page.Params.license }}
  • {{ T "licenseLabel" }} {{ . }}
    • {{- end -}}
{{- end -}}
